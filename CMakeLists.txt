cmake_minimum_required(VERSION 3.24)
project(HelloFace VERSION 0.1.0 LANGUAGES CXX)
option(HELLOFACE_BUILD_TESTS "Build portable protocol tests" ON)
option(HELLOFACE_BUILD_WINDOWS_COMPONENTS "Build Windows-only components (not enabled until audited)" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(helloface_ipc INTERFACE)
target_include_directories(helloface_ipc INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(helloface_ipc INTERFACE cxx_std_20)
if(MSVC)
  add_compile_options(/W4 /permissive- /EHsc /utf-8 /guard:cf /sdl)
  add_link_options(/guard:cf /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
if(HELLOFACE_BUILD_TESTS)
  enable_testing()
  add_executable(helloface_protocol_tests tests/protocol_tests.cpp)
  target_link_libraries(helloface_protocol_tests PRIVATE helloface_ipc)
  add_test(NAME helloface_protocol_tests COMMAND helloface_protocol_tests)
endif()
if(WIN32)
  add_executable(helloface_config WIN32 src/config/main.cpp)
  target_compile_definitions(helloface_config PRIVATE UNICODE _UNICODE)
  target_link_libraries(helloface_config PRIVATE user32 gdi32)
  set_target_properties(helloface_config PROPERTIES OUTPUT_NAME "HelloFace")
endif()

if(WIN32 AND HELLOFACE_BUILD_WINDOWS_COMPONENTS)
  message(FATAL_ERROR "Windows authentication components remain gated pending security review. See docs/IMPLEMENTATION_PLAN.md.")
endif()
